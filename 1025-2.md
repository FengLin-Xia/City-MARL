# 工业集群解锁机制设计文档 v5.0

## 📋 项目概述

### 目标
在现有v5.0架构基础上，实现工业集群解锁机制，类似RTS游戏中的科技树系统。通过前置条件解锁高级工业动作，并实现全局协同效应。

### 核心特性
- **解锁机制**：需要建造20个传统工业建筑（IND 3/4/5）才能解锁高级工业动作（IND 9/10/11）
- **协同效应**：高级工业建筑建造后，全局提升传统工业建筑的奖励
- **Hub3专用**：高级工业动作只能在Hub3候选范围内建造
- **配置驱动**：所有参数通过配置文件控制，保持v5.0架构

```python
def _register_building(self, agent: str, action_id: int, slots: List[str]) -> None:
    """注册新建筑到注册表"""
    building_id = f"{agent}_{action_id}_{self.current_month}_{len(self.building_registry.buildings)}"
    
    # 获取建筑位置
    position = self._get_building_position(slots)
    
    # 创建建筑信息
    building_info = BuildingInfo(
        building_id=building_id,
        action_id=action_id,
        agent=agent,
        slots=slots,
        position=position,
        build_month=self.current_month
    )
    
    # 注册建筑
    self.building_registry.buildings[building_id] = building_info
    self.building_registry.action_counts[action_id] = self.building_registry.action_counts.get(action_id, 0) + 1
    
    # 检查解锁条件
    self._check_unlock_conditions()
    
    # 检查协同效应
    self._check_cluster_bonus()

def _check_unlock_conditions(self) -> None:
    """检查解锁条件"""
    for action_id in [9, 10, 11]:
        if self.unlock_status.get(action_id, False):
            continue
            
        config = self.config.get("unlock_conditions", {}).get(str(action_id), {})
        prerequisite_actions = config.get("prerequisite_actions", [])
        min_count = config.get("min_count", 0)
        
        # 统计前置动作数量
        total_count = sum(self.building_registry.action_counts.get(aid, 0) for aid in prerequisite_actions)
        
        if total_count >= min_count:
            self.unlock_status[action_id] = True
            self.logger.info(f"动作 {action_id} 已解锁！前置条件：{total_count}/{min_count}")

def _check_cluster_bonus(self) -> None:
    """检查协同效应激活"""
    if self.cluster_bonus_active:
        return
        
    # 检查是否有高级工业建筑
    advanced_count = sum(self.building_registry.action_counts.get(aid, 0) for aid in [9, 10, 11])
    
    if advanced_count > 0:
        self.cluster_bonus_active = True
        self.logger.info(f"工业集群协同效应已激活！高级工业建筑数量：{advanced_count}")

def _get_building_position(self, slots: List[str]) -> Tuple[float, float]:
    """获取建筑位置（取中心点）"""
    if not slots:
        return (0.0, 0.0)
    
    positions = []
    for slot_id in slots:
        slot_info = self.enumerator.slots.get(slot_id)
        if slot_info:
            positions.append((slot_info.x, slot_info.y))
    
    if not positions:
        return (0.0, 0.0)
    
    # 计算中心点
    center_x = sum(pos[0] for pos in positions) / len(positions)
    center_y = sum(pos[1] for pos in positions) / len(positions)
    
    return (center_x, center_y)
```

### 3. 协同奖励模块 (`reward_terms/industrial_cluster_reward.py`)

```python
"""
工业集群协同奖励模块

实现高级工业建筑对传统工业建筑的全局奖励加成。
"""

from typing import Dict, Any
from contracts import EnvironmentState


class IndustrialClusterRewardTerm:
    """工业集群协同奖励项"""
    
    def __init__(self, config: Dict[str, Any]):
        self.config = config.get("reward_mechanisms", {}).get("industrial_cluster_bonus", {})
        self.enabled = self.config.get("enabled", False)
        self.bonus_rate = self.config.get("bonus_rate", 0.15)
        self.affects_actions = self.config.get("affects_actions", [3, 4, 5])
        self.trigger_actions = self.config.get("trigger_actions", [9, 10, 11])
        self.global_effect = self.config.get("global_effect", True)
    
    def compute(self, prev_state: EnvironmentState, state: EnvironmentState, action_id: int) -> float:
        """
        计算工业集群协同奖励
        
        Args:
            prev_state: 前一个状态
            state: 当前状态
            action_id: 动作ID
            
        Returns:
            协同奖励值
        """
        if not self.enabled:
            return 0.0
        
        # 只对传统工业建筑计算协同奖励
        if action_id not in self.affects_actions:
            return 0.0
        
        # 检查是否有高级工业建筑激活协同效应
        if not self._is_cluster_bonus_active(state):
            return 0.0
        
        # 计算基础奖励
        base_reward = self._get_base_reward(action_id)
        
        # 应用协同加成
        cluster_bonus = base_reward * self.bonus_rate
        
        return cluster_bonus
    
    def _is_cluster_bonus_active(self, state: EnvironmentState) -> bool:
        """检查协同效应是否激活"""
        # 从状态中获取建筑注册表信息
        if hasattr(state, 'building_registry'):
            advanced_count = sum(
                state.building_registry.action_counts.get(aid, 0) 
                for aid in self.trigger_actions
            )
            return advanced_count > 0
        
        return False
    
    def _get_base_reward(self, action_id: int) -> float:
        """获取基础奖励值"""
        # 从配置中获取基础奖励
        action_params = self.config.get("action_params", {}).get(str(action_id), {})
        return action_params.get("base_reward", 0.0)
```
